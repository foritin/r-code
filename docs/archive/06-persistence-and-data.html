持久化与数据模型

1. 概述

持久层封装在 persistence crate，接口不暴露底层 SQLite 类型，
必要时可换实现。

2. 数据库选型

2.1 当前实现

  • node:sqlite（Node 24 内建，Electron 43 验证可用）。
  • 同步 API + WAL + 事务。

2.2 Rust 替代

  • rusqlite — 同步 SQLite 访问。
  • sqlx — 异步 SQLite 访问（与 Tokio 集成更好）。

3. 核心数据模型

3.1 任务 (tasks)

sql
CREATE TABLE tasks (
    id TEXT PRIMARY KEY,
    project_id TEXT NOT NULL,
    title TEXT NOT NULL,
    goal TEXT,
    mode TEXT NOT NULL,  -- ask | edit | auto
    state TEXT NOT NULL, -- idle | exploring | in_progress | review_ready | archived
    archived INTEGER DEFAULT 0,
    worktree_path TEXT,
    created_at INTEGER NOT NULL,
    updated_at INTEGER NOT NULL
);

3.2 轮次 (agent_runs)

sql
CREATE TABLE agent_runs (
    id TEXT PRIMARY KEY,
    task_id TEXT NOT NULL REFERENCES tasks(id),
    model TEXT,
    review_state TEXT NOT NULL, -- pending | accepted | auto_accepted | rolled_back | answered
    reviewed_at INTEGER,
    started_at INTEGER NOT NULL,
    ended_at INTEGER,
    usage_json TEXT  -- token usage, cost, etc.
);

3.3 工具调用 (tool_calls)

sql
CREATE TABLE tool_calls (
    id TEXT PRIMARY KEY,
    run_id TEXT NOT NULL REFERENCES agent_runs(id),
    task_id TEXT NOT NULL REFERENCES tasks(id),
    tool_name TEXT NOT NULL,
    input_json TEXT NOT NULL,
    output_json TEXT,
    risk_level TEXT NOT NULL,
    status TEXT NOT NULL, -- running | ok | error | denied
    started_at INTEGER NOT NULL,
    ended_at INTEGER
);

3.4 文件变更 (file_changes)

sql
CREATE TABLE file_changes (
    id TEXT PRIMARY KEY,
    task_id TEXT NOT NULL REFERENCES tasks(id),
    tool_call_id TEXT REFERENCES tool_calls(id),
    path TEXT NOT NULL,
    kind TEXT NOT NULL, -- create | modify | delete | rename
    before_hash TEXT,
    after_hash TEXT,
    created_at INTEGER NOT NULL
);

3.5 文件基线 (file_baselines)

sql
CREATE TABLE file_baselines (
    id TEXT PRIMARY KEY,
    task_id TEXT NOT NULL REFERENCES tasks(id),
    path TEXT NOT NULL,
    content_hash TEXT NOT NULL,
    blob_id TEXT NOT NULL,
    captured_at INTEGER NOT NULL
);

3.6 内容寻址 Blob (blobs)

sql
CREATE TABLE blobs (
    id TEXT PRIMARY KEY,  -- content hash
    size INTEGER NOT NULL,
    ref_count INTEGER NOT NULL DEFAULT 1,
    created_at INTEGER NOT NULL
);

3.7 权限请求 (permission_requests)

sql
CREATE TABLE permission_requests (
    id TEXT PRIMARY KEY,
    task_id TEXT NOT NULL REFERENCES tasks(id),
    tool_call_id TEXT REFERENCES tool_calls(id),
    tool_name TEXT NOT NULL,
    risk_level TEXT NOT NULL,
    status TEXT NOT NULL, -- pending | approved | denied
    resolved_at INTEGER,
    created_at INTEGER NOT NULL
);

3.8 工作空间 (workspaces)

sql
CREATE TABLE workspaces (
    id TEXT PRIMARY KEY,
    canonical_path TEXT NOT NULL UNIQUE,
    display_name TEXT NOT NULL,
    trust_state TEXT NOT NULL, -- untrusted | trusted
    last_opened_at INTEGER,
    created_at INTEGER NOT NULL
);

3.9 任务事件 (task_events)

sql
CREATE TABLE task_events (
    id TEXT PRIMARY KEY,
    task_id TEXT NOT NULL REFERENCES tasks(id),
    type TEXT NOT NULL,
    data_json TEXT NOT NULL,
    created_at INTEGER NOT NULL
);

4. 事务与备份

4.1 事务

  • 所有写操作走事务。
  • WAL 模式支持并发读。

4.2 备份

  • 定期备份到 userData 目录。
  • 校验和验证。
  • 恢复机制。

6. 记账策略

6.1 全量记录

  • 所有工具调用入 tool_calls。
  • 调用者身份（task/session id 或 terminal:<id>）。

6.2 输出处理

  • send 注入文本完整记录（已过分类）。
  • read/wait 只记元数据。
  • 捕获的输出不持久化。

6.3 滚动缓冲

  • 每终端 ANSI-stripped 滚动缓冲 ~200 KB。
  • 仅内存，随终端消亡。

7. 搜索索引

7.1 全文搜索

  • 任务标题、目标。
  • 文件路径。

7.2 索引策略

  • SQLite FTS5。
  • 或 tantivy 独立索引。

8. Rust 实现要点

// 数据库连接
struct Database {
    conn: rusqlite::Connection,
}

impl Database {
    fn open(path: &Path) -> Result<Self, DbError> {
        let conn = rusqlite::Connection::open(path)?;
        conn.pragma_update(None, "journal_mode", "WAL")?;
        conn.pragma_update(None, "foreign_keys", "ON")?;
        Ok(Self { conn })
    }
    
    fn migrate(&self) -> Result<(), DbError> { ... }
}

// Repository 模式
struct TaskRepository {
    db: Arc<Database>,
}

impl TaskRepository {

    fn get(&self, id: &str) -> Result<Option<Task>, DbError> { ... }
    fn list(&self, scope: TaskScope) -> Result<Vec<Task>, DbError> { ... }
    fn update_state(&self, id: &str, state: TaskState) -> Result<(), DbError> { ... }
}

// 内容寻址 Blob
struct BlobStore {
    db: Arc<Database>,
    blobs_dir: PathBuf,
}

impl BlobStore {
    fn put(&self, content: &[u8]) -> Result<String, BlobError> {
        let hash = blake3::hash(content).to_hex().to_string();
        // dedup: if hash exists, increment ref_count
        // else write to disk
        Ok(hash)
    }
    
    fn get(&self, hash: &str) -> Result<Vec<u8>, BlobError> { ... }
}
